#include<iostream>
#include<memory>
struct node {
	int val;
	std::unique_ptr<node>left;
	std::unique_ptr<node>right;
	int h;
	node(int val) :val(val), h(1) {}
};
int geth(const std::unique_ptr<node>& root) {
	if (!root)return 0;
	return root->h;
}
int diff_h(const std::unique_ptr<node>& a, const std::unique_ptr<node>& b) {
	return geth(a) - geth(b);
}
void LL(std::unique_ptr<node>& x) {
	std::unique_ptr<node>y = std::move(x->left);
	x->left = std::move(y->right);
	y->right = std::move(x);
	x = std::move(y);
	x->right->h = std::max(geth(x->right->left), geth(x->right->right)) + 1;
	x->h = std::max(geth(x->left), geth(x->right)) + 1;
}
void RR(std::unique_ptr<node>& x) {
	std::unique_ptr<node>y = std::move(x->right);
	x->right = std::move(y->left);
	y->left = std::move(x);
	x = std::move(y);
	x->left->h = std::max(geth(x->left->left), geth(x->left->right)) + 1;
	x->h = std::max(geth(x->left), geth(x->right)) + 1;
}
void LR(std::unique_ptr<node>& x) {
	RR(x->left);
	LL(x);
}
void RL(std::unique_ptr<node>& x) {
	LL(x->right);
	RR(x);
}
void insert(std::unique_ptr<node>& root, int val) {
	if (!root) {
		root = std::make_unique<node>(val);
		return;
	}
	if (val < root->val) {
		insert(root->left, val);
		if (diff_h(root->left, root->right) > 1) {
			node* p = root->left.get();
			if (diff_h(p->left, p->right) >= 1) {
				LL(root);
			}
			else {
				LR(root);
			}
		}
	}
	else if (val > root->val) {
		insert(root->right, val);
		if (diff_h(root->right, root->left) > 1) {
			node* p =root->right.get();
			if (diff_h(p->right, p->left) >= 1) {
				RR(root);
			}
			else {
				RL(root);
			}
		}
	}
	else return;
	root->h = std::max(geth(root->left), geth(root->right)) + 1;
}
void del(std::unique_ptr<node>& root, int val) {
	if (!root)return;
	if (val < root->val) {
		del(root->left, val);
		if (diff_h(root->right, root->left) > 1) {
			node* p = root->right.get();
			if (geth(p->right) >= geth(p->left)) {
				RR(root);
			}
			else {
				RL(root);
			}
		}
	}
	else if (val > root->val) {
		del(root->right, val);
		if (diff_h(root->left, root->right) > 1) {
			node* p = root->left.get();
			if (geth(p->left) >= geth(p->right)) {
				LL(root);
			}
			else {
				LR(root);
			}
		}
	}
	else {
		if (!root->left) {
			root = std::move(root->right);
		}
		else if (!root->right) {
			root = std::move(root->left);
		}
		else {
			node* p = root->left.get();
			while (p->right) {
				p = p->right.get();
			}
			root->val = p->val;
			del(root->left, p->val);
			if (diff_h(root->right,root->left) > 1) {
				node* temp = root->right.get();
				if (geth(temp->right) >= geth(temp->left)) {
					RR(root);
				}
				else {
					RL(root);
				}
			}
			
		}
	}
	if(root)root->h = std::max(geth(root->left), geth(root->right)) + 1;

}
void in_order(const node* root) {
	if (!root)return;
	in_order(root->left.get());
	std::cout << root->val << ' ';
	in_order(root->right.get());
}

先附上代码。

首先困扰很久的肯定就是del的时候，为什么子树的左右子树高度相等的时候要用单旋。其实**双旋也可以**，只不过指针会多变换几次，效率低了点而已，但是**双旋是不会出错的**。

还有一点就是之前一直不太理解失衡之后调整的时机，其实很容易理解，就是从递归的最深处开始往回一步步调整(也很容易看到每次调整的时候都是在插入或者删除之后，所以**旋转也是个回溯的状态**) 

为什么**插入**的时候不会出现子树的左右子树等高的情况？(已知失衡)因为既然已经知道失衡了，而且还是插入，必然是有一边子树是变高的，才会导致上面失衡，所以子树的左右子树高度不可能相等。而且**插入的时候，如果要旋转，最多旋转一次就好了**，因为在失衡的节点，总树高调整之前和调整之后是一样的，不会影响祖先节点。(这个自己手动模拟一个个节点插进去，之后计算前后高度就行了，是成立的)而这个时候就引申出了为什么**删除的时候可能会回溯往上旋转很多次**---因为删除的时候，如果出现了失衡，调整过后，这棵子树的树高**可能**是会减少的，如果这个子树树高减少了，那就有可能会一直影响到上面的祖父节点。(可能也不会减少，就比如说子树的左右子树 树高相同的时候 旋转一下 其实那棵子树的树高就不会减少)
