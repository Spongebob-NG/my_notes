/*一定要注意 unique_ptr没有拷贝构造函数 虽然能值传递 但是中间涉及move 容易出错
如果要对树进行修改操作的话 用unique_ptr 
如果只是遍历或者观察的话 用普通的指针就好了(注意要.get())*/

//引用传参
void insert_dfs(std::unique_ptr<node>&root, int val) {
	if (!root) {
		root = std::make_unique<node>(val);
		return;
	}
	if (val < root->val) {
		insert_dfs(root->left, val);
	}
	else if (val > root->val) {
		insert_dfs(root->right, val);
	}
	else return;//相等的话就没必要再插入了 同样的节点只能插入一次
}
//值传参
std::unique_ptr<node>insert_dfs2(std::unique_ptr<node>root, int val) {
	if (!root) {
		return std::make_unique<node>(val);
	}
	if (val < root->val) {
		root->left = insert_dfs2(std::move(root->left), val);
	}
	else if (val > root->val) {
		root->right = insert_dfs2(std::move(root->right), val);
	}
	return root;
}

void insert(std::unique_ptr<node>& root, int val) {
	if (!root) {
		root = std::make_unique<node>(val);
		return;
	}
	node* p = root.get();
	node* parent = nullptr;
	while (p) {
		parent = p;
		if (val < parent->val) {
			p = p->left.get();
		}
		else if (val > parent->val) {
			p = p->right.get();
		}
		else return;
	}
	if (val < parent->val) {
		parent->left = std::make_unique<node>(val);
	}
	else {
		parent->right = std::make_unique<node>(val);
	}
}
node* find(std::unique_ptr<node>& root, int val) {
	if (!root || root->val == val)return root.get();
	if (val < root->val)return find(root->left, val);
	else if (val > root->val)return find(root->right, val);
}
//引用传递删除
void del(std::unique_ptr<node>& root, int val) {
	if (!root)return;
	if (root->val == val) {
		if (!root->left) {
			root = std::move(root->right);
		}
		else if (!root->right) {
			root = std::move(root->left);
		}
		else {
			node* p = root->left.get();
			while (p->right) {
				p = p->right.get();
			}
			root->val = p->val;
			del(root->left, p->val);
		}
	}
	else if (val < root->val) {
		del(root->left, val);
	}
	else {
		del(root->right, val);
	}
}
//值传递删除
std::unique_ptr<node>del2(std::unique_ptr<node>root, int val) {
	if (!root)return root;
	if (root->val == val) {
		if (!root->left) {
			root = std::move(root->right);
		}
		else if (!root->right) {
			root = std::move(root->left);
		}
		else {
			node* p = root->left.get();
			while (p->right) {
				p = p->right.get();
			}
			root->val = p->val;
			root->left = del2(std::move(root->left), p->val);
		}
	}
	else if (val < root->val) {
		root->left = del2(std::move(root->left), val);
	}
	else root->right = del2(std::move(root->right), val);
	return root;
}
