int lower_bound_manual(vector<int>& v, int x) {
    int l = 0, r = v.size(); // 注意r不是size()-1，而是size()
    while (l < r) {
        int mid = (l + r) / 2;
        if (v[mid] < x)
            l = mid + 1;   // 中间值小于x，目标在右边
        else
            r = mid;       // 否则收缩右边界
    }
    return l; // l就是第一个≥x的位置
}
