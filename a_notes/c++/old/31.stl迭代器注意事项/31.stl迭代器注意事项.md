```c++
class Solution {
public:
    int find_left(vector<int>& nums, int target) {
        auto it = lower_bound(nums.begin(), nums.end(), target);
        if (it != nums.end()&&*it==target)return it - nums.begin();
        else return -1;
    }
    int find_right(vector<int>& nums, int target) {
        auto it = lower_bound(nums.begin(), nums.end(), target + 1);
        if (it != nums.begin() && *(it - 1) == target)return it - 1 - nums.begin();
        else return -1;
    }
    vector<int> searchRange(vector<int>& nums, int target) {
        vector<int>ans;
        ans.push_back(find_left(nums, target));
        ans.push_back(find_right(nums, target));
        return ans;
    }
};
```
