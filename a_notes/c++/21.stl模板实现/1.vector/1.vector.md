#include<iostream>
#include<stdio.h>
#include<string.h>
#include<map>
#include<unordered_map>
#include<queue>
#include<deque>
#include<stack>
#include<algorithm>
using namespace std;
template<typename T>
class my_vector {
public:
	my_vector() :start(NULL), finish(NULL), end_of_storage(NULL) {}
	//可以直接都赋值为NULL，因为NULL-NULL=0，不会影响后面的判断
	my_vector(size_t n) {
		start = new T[n];
		finish = start;
		end_of_storage = start + n;
		//i的类型最好要是size_t，因为i肯定是正整数，这样的话n非常大的时候，能比int数值大一倍
		for (size_t i = 0; i < n; i++) {
			start[i] = T();//T()是默认值，一般都是0
		}
	}
	my_vector(size_t n, const T&value) {
		start = new T[n];
		end_of_storage = start + n;
		finish = start + n;
		for (size_t i = 0; i < n; i++) {
			start[i] = value;
		}
	}
	//复制构造函数，都是这么写的，你记住就行
	my_vector(const my_vector& vec) {
		size_t size = vec.finish - vec.start;
		size_t capacity = vec.end_of_storage - vec.start;
		start = new T[capacity];
		for (int i = 0; i < size; i++) {
			start[i] = vec.start[i];
		}
		finish = start + size;
		end_of_storage = start + capacity;
	}
	//重定义=，跟上面那个有点区别，上面那个是定义的时候就赋值，这个是定义了之后先不赋值，后面再赋值也可以
	/*赋值运算符一定要返回引用！！不返回的话，如果代码稍微复杂一点，如果是链式赋值，比如a = b = c的话，b = c之后创建的
	是一个临时副本，销毁的时候会调用析构函数两次，绝对会出问题！
	而且！！c++标准里面也明确强调了赋值运算符必须返回引用
	*/
	my_vector& operator=(const my_vector& vec) {
		//要用&vec！这里不是引用了，这里是取地址的意思，别学多了搞混了最基本的了
		if (this != &vec) {
			delete[]start;
			size_t size = vec.finish - vec.start;
			size_t capacity = vec.end_of_storage - vec.start;
			start = new T[capacity];
			for (int i = 0; i < size; i++) {
				start[i] = vec.start[i];
			}
			finish = start + size;
			end_of_storage = start + capacity;
		}
		return *this;
		//this本质上还是个指针，所以返回的是*this
	}
	void reserve(size_t new_capacity) {
		size_t old_size = finish - start;
		size_t old_capacity = end_of_storage - start;
		if (new_capacity > old_capacity) {
			T* newdata = new T[new_capacity];
			for (int i = 0; i < old_size; i++) {
				newdata[i] = start[i];
			}
			delete[]start;
			start = newdata;
			finish = start + old_size;
			end_of_storage = start + new_capacity;
		}
	}
	//这里也是返回引用，因为传参的时候传的直接就是本身，而不是又开销一个副本。如果参数类型比较大的话就会很废内存
	void resize(size_t n, const T& value = T()) {
		size_t old_size = finish - start;
		if (n > end_of_storage) {
			reserve(n);
		}
		if (n > old_size) {
			for (int i = old_size; i < n; i++) {
				start[i] = value;
			}
		}
		finish = start + n;
	}
	void clear() {
		finish = start;
	}
	void push_back(const T& value) {
		if (finish == end_of_storage) {
			reallocate();
		}
		*finish = value;
		finish++;
	}
	void order() {
		for (T* i = start; i != finish; i++) {
			cout << (*i) << ' ';
		}
	}
	//这里等复习了错误之后其实可以抛出错误
	T& operator[](size_t n) {
		return start[n];
	}
	~my_vector() {
		delete[]start;
		start = finish = end_of_storage = NULL;
	}
private:
	T* start;
	T* finish;
	T* end_of_storage;
	void reallocate() {
		size_t old_size = finish - start;
		size_t old_capacity = end_of_storage - start;
		size_t new_capacity = old_capacity == 0 ? 1 : old_capacity * 2;
		T* newdata = new T[new_capacity];
		for (size_t i = 0; i < old_size; i++) {
			newdata[i] = start[i];
		}
		delete[]start;
		start = newdata;
		finish = start + old_size;
		end_of_storage = start + new_capacity;
	}
};
int main(int argc, const char* argv[]) {
	my_vector<int>vec;
	vec.push_back(1);
	vec.push_back(1);
	vec.push_back(1);
	vec.order();
	cout << endl;
	my_vector<int>vec2 = vec;
	vec2.order();
	return 0;
}
