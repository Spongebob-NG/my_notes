这题其实还行 一开始以为所有同学必须都得拔河，所以用了前缀和 但是不是所有同学都要去拔河的。而这种很明显地分左右区间的，就需要设个中间分界线，刚好左边出现过的sum都可以用set来维护，右边的sum直接一个个求和，再用lower_bound看左边有没有出现在sum附近的，然后更新ans即可

```c++
#include<iostream>
#include<set>
#include<climits>
using namespace std;
typedef long long ll;
const int N = 1005;
ll a[N];
int n;
set<ll>mp;
int main() {
	cin >> n;
	for (int i = 1; i <= n; i++)cin >> a[i];
	ll ans = INT_MAX;
	for (int mid = 2; mid <= n; mid++) {
		ll sum = 0;
		for (int l = mid - 1; l >= 1; l--) {
			sum += a[l];
			mp.insert(sum);
		}
		sum = 0;
		for (int r = mid; r <= n; r++) {
			sum += a[r];
			auto it = mp.lower_bound(sum);
			if (it != mp.end())ans = min(ans, *it - sum);
			if (it != mp.begin())ans = min(ans, sum - *--it);
		}
	}
	cout << ans << endl;
}
```
