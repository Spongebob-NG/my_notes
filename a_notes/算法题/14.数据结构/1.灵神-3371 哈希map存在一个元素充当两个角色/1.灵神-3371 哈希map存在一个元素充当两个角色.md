```c++
class Solution {
public:
    int getLargestOutlier(vector<int>& nums) {
        unordered_map<int, int>cnt;
        long long sum = 0;
        for (auto x : nums)sum += x;
        for (int i = 0; i < nums.size(); i++) {
            cnt[nums[i]]++;
        }
        int ans = -1005;
        for (int i = 0; i < nums.size(); i++) {
            auto it = cnt.find(sum - 2 * nums[i]);
            if (it == cnt.end()||(it->first==nums[i]&&it->second<=1))continue;
            ans = max(ans, it->first);
        }
        return ans;
    }
};
```

从这一题开始长了个记性：这种哈希表的题目，有可能目标值和当前元素相等，而且不能把这个值分开成两个身份。如果相等的话必须要满足个数大于1，否则的话不成立
