```c++
#include<iostream>
using namespace std;
const int N = 100005;
int a[N];
int ans[N];//避免跟原来数组混起来 干脆直接来个ans数组存答案
int main() {
	int n; string d;
	cin >> n >> d;
	int len = d.length();
	int pos = d.find('.');//直接用find找小数点的位置 效率更高
	int post_len = len - 1 - pos;
	int cnt = 0;
	for (int i = d.length() - 1; i >= 0; i--) {
		if (d[i] == '.')continue;
		a[cnt++] = d[i]-'0';
	}//循环结束后cnt刚好是看作大整数时的长度

	for (int i = 0; i < n; i++) {
		int t = 0;
		for (int j = 0; j < cnt; j++) {
			t += a[j] * 2;
			a[j] = t % 10;
			t /= 10;
		}
		if (t > 0)a[cnt++] = t;
	}//每一步都必须要进位 如果最后才进位的话，中间的结果可能已经溢出了
	int carry = 0;
	if (a[post_len - 1] >= 5) {
		carry = 1;
	}
	int ans_len = 0;
	int t = carry;
	for (int i = post_len; i < cnt; i++) {
		t += a[i];
		ans[ans_len++] = t % 10;
		t /= 10;
	}
	if (t > 0)ans[ans_len++] = t;
	while (ans_len > 0 && ans[ans_len] == 0) {
		ans_len--;
	}
	for (int i = ans_len; i >= 0; i--)cout << ans[i];
	}
	

```

这道题是高精度乘法题 如果不用高精度的话只能通过百分之20的数据 具体思路如何 看代码和注释即可
