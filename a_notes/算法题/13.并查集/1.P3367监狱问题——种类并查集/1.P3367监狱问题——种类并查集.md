#### P3367监狱问题——**种类并查集**

```c++

#include<iostream>
#include<stdio.h>
#include<string.h>
#include<map>
#include<unordered_map>
#include<queue>
#include<deque>
#include<stack>
#include<algorithm>
using namespace std;
int f[40005];
int find(int i) {
	if (f[i] == i)return i;
	return f[i] = find(f[i]);
}
struct prime {
	int x, y, c;
}a[100005];
bool cmp(prime& a, prime& b) {
	return a.c > b.c;
}
int n; int m;
int main(int argc, const char* argv[]) {
	cin >> n >> m;
	int x, y, fx, fy, fxd, fyd;
	for (int i = 1; i <= n; i++) {
		f[i] = i;
		f[i + n] = i + n;
	}
	for (int i = 0; i < m; i++) {
		cin >> a[i].x >> a[i].y >> a[i].c;
	}
	sort(a, a + m,cmp);
	for (int i = 0; i < m; i++) {
		x = a[i].x;
		y = a[i].y;
		fx = find(x);
		fy = find(y);
		if (fx == fy) {
			cout << a[i].c << endl;
			return 0;
		}
		fxd = find(x + n);
		fyd = find(y + n);
		f[fx] = fyd;
		f[fy] = fxd;
	}
	cout << 0 << endl;
	return 0;
}

```

怎么理解种类并查集？

为什么会出现“敌对关系”？因为并查集只能表示同一个集合，但是不能表示不同集合，所以需要假想一个敌人才行，根据“敌人的敌人是朋友”原则来建立不同的关系
