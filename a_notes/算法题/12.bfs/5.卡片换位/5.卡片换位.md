```c++
#include <iostream>
#include <queue>
#include <vector>
#include <set> // 补丁：必须用 set 来查重
#include <string>

using namespace std;

// 补丁：改名避免和节点名冲突
vector<vector<char>> m_global(2, vector<char>(3));
int ai, aj, bi, bj;

struct node {
	vector<vector<char>> map;
	int cur_ai, cur_aj, cur_bi, cur_bj;
	int step;
};

// 补丁：去掉原地不动的 {0, 0}
int dx[4] = { 0, 0, 1, -1 };
int dy[4] = { 1, -1, 0, 0 };

void bfs() {
	queue<node> q;
	set<vector<vector<char>>> vis; // 记录访问过的二维地图

	node start_node;
	start_node.map = m_global; // 把真实的地图装进去
	start_node.cur_ai = ai; start_node.cur_aj = aj; 
	start_node.cur_bi = bi; start_node.cur_bj = bj;
	start_node.step = 0;
	
	q.push(start_node);
	vis.insert(start_node.map); // 起点打上标记

	while (!q.empty()) {
		node head = q.front();
		q.pop();
		vector<vector<char>> temp = head.map;

		// 你的硬核思路：遍历全图找卡片
		for (int i = 0; i < 2; i++) {
			for (int j = 0; j < 3; j++) {
				if (temp[i][j] != ' ') { // 找到一张实体卡片
				
					for (int cnt = 0; cnt < 4; cnt++) {
						int xi = i + dx[cnt];
						int yi = j + dy[cnt];
						if (xi < 0 || xi >= 2 || yi < 0 || yi >= 3) continue;
						
						// 如果卡片旁边刚好是空格，就发生交换
						if (temp[xi][yi] == ' ') {
							vector<vector<char>> temp1 = temp;
							
							// 交换：现在 temp1[xi][yi] 是卡片了，temp1[i][j] 是空格了
							char a = temp1[i][j];
							temp1[i][j] = temp1[xi][yi];
							temp1[xi][yi] = a;

							// 补丁：如果这个地图以前换出来过，绝对不走回头路！
							if (vis.count(temp1)) continue;

							// 补丁：先继承原来的所有坐标！防止失忆
							int next_ai = head.cur_ai, next_aj = head.cur_aj;
							int next_bi = head.cur_bi, next_bj = head.cur_bj;

							// 看看这次移动的是不是 A 或者 B
							if (temp1[xi][yi] == 'A') {
								next_ai = xi;
								next_aj = yi;
							}
							if (temp1[xi][yi] == 'B') {
								next_bi = xi;
								next_bj = yi;
							}

							// 终点核对
							if (next_ai == bi && next_aj == bj && next_bi == ai && next_bj == aj) {
								cout << head.step + 1 << endl;
								return; // bfs 结束
							}

							q.push({ temp1, next_ai, next_aj, next_bi, next_bj, head.step + 1 });
							vis.insert(temp1); // 标记新地图
						}
					}
				}
			}
		}
	}
}

int main() {
	// 补丁：因为有真实的空格字符，用 cin >> 读不到，必须整行读
	string s1, s2;
	getline(cin, s1);
	getline(cin, s2);
	
	for(int j=0; j<3; j++) m_global[0][j] = s1[j];
	for(int j=0; j<3; j++) m_global[1][j] = s2[j];

	for (int i = 0; i < 2; i++) {
		for (int j = 0; j < 3; j++) {
			// 题目标准是大写 A 和 B
			if (m_global[i][j] == 'A') {
				ai = i; aj = j;
			}
			if (m_global[i][j] == 'B') {
				bi = i; bj = j;
			}
		}
	}
	bfs();
	return 0;
}
```

思路方面，这道题一开始方向是想对了，但是有几个细节没把握好：

1.全局变量和局部变量名字冲突，这个是老毛病了

2.cin会跳过空格符，不能用cin输入字符数组，否则吃不到空格，要用getline（注意，只能geline放到s里面，再循环赋值，不能直接m_global[i]=getline(cin,s)（×），两边类型完全不一样

3.遍历 一定一定要有一个标记数组！不然会一直重复，无限重复，尤其是甚至一开始把dx=0,dy=0也写入了距离数组中



**更方便的解法在这：**(尤其是针对整个图中只有一个空格的这种类似题，那个唯一的才是主角，上面那个方法思路上太复杂太麻烦了，每个点每一轮都要动，不如直接只动空格)

```c++
#include <iostream>
#include <queue>
#include <unordered_map>
#include <string>

using namespace std;

// 结构体只存当前状态和走到这一步花的步数
struct node {
    string s;
    int step;
};

// 只有上下左右 4 个方向，不需要自己待在原地不动
int dx[4] = {0, 0, 1, -1};
int dy[4] = {1, -1, 0, 0};

void bfs(string start) {
    int pos_A = -1, pos_B = -1;
    
    // 1. 找到初始状态下 A 和 B 的一维下标
    for (int i = 0; i < 6; i++) {
        if (start[i] == 'A') pos_A = i;
        if (start[i] == 'B') pos_B = i;
    }

    queue<node> q;
    unordered_map<string, int> vis; // 绝对不能忘的查重神器

    q.push({start, 0});
    vis[start] = 1;

    while (!q.empty()) {
        string curr = q.front().s;
        int step = q.front().step;
        q.pop();

        // 2. 终点判定：原来 A 的位置变成了 B，原来 B 的位置变成了 A
        if (curr[pos_A] == 'B' && curr[pos_B] == 'A') {
            cout << step << endl;
            return;
        }

        // 3. 寻找唯一的“主角”：空格的一维下标
        int space_idx = curr.find(' ');
        
        // 核心坐标转换：一维下标转二维坐标 (x, y)
        // 对于 2行3列 的网格：行 = idx / 列数，列 = idx % 列数
        int x = space_idx / 3;
        int y = space_idx % 3;

        // 4. 让空格向四个方向移动
        for (int i = 0; i < 4; i++) {
            int nx = x + dx[i];
            int ny = y + dy[i];

            // 越界检查
            if (nx < 0 || nx >= 2 || ny < 0 || ny >= 3) continue;

            // 核心坐标转换：二维坐标 (nx, ny) 转变回一维下标
            int next_idx = nx * 3 + ny;
            
            string next_s = curr;
            // 交换空格和目标卡片的位置
            swap(next_s[space_idx], next_s[next_idx]);

            // 如果这个新排布以前没见过，就入队
            if (vis[next_s] == 0) {
                vis[next_s] = 1;
                q.push({next_s, step + 1});
            }
        }
    }
}

int main() {
    string s1, s2;
    // 应对带有空格的输入，必须用 getline
    getline(cin, s1);
    getline(cin, s2);
    
    // 将两行直接拼成一个长度为 6 的字符串
    string start = s1 + s2; 
    
    bfs(start);

    return 0;
}
```
