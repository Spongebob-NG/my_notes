```c++
#include <iostream>
using namespace std;
int tower[505][505];
int choice[2] = { 0,1 };
int m, n;
int k;
int ans = 0;
void dfs(int cur,int num_a,int num_b) {
	if (num_a > m || num_b > n)return;
	if (cur == k) {
		ans++;
		return;
	}
	for (int i = 0; i <= 1; i++) {
		int ta = 0, tb = 0;
		int select = choice[i];
		tower[0][cur] = select;
		if (select == 0)ta=1; else tb=1;
		for (int i = 1; i <= cur; i++) {
			tower[i][cur] = (tower[i - 1][cur] == tower[i - 1][cur -1] ? 0 : 1);
			if (tower[i][cur] == 0) ta++; else tb++;
		}
		dfs(cur+1, num_a+ta, num_b+tb);
	}
}
int main()
{
	cin >> m >> n;
	int sum = 0;
	for (int i = 1; i <= m + n; i++) {
		sum += i;
		if (sum == n + m) {
			k = i;
			break;
		}
	}
	for (int i = 0; i <= 1; i++) {
		int select = choice[i];
		if (select == 0) {
			tower[0][0] = 0;
			dfs(1, 1, 0);
		}
		else {
			tower[0][0] = 1;
			dfs(1, 0, 1);
		}
	}
	cout << ans << endl;


	return 0;
}
```

这是机器人塔问题

第一次接触这种的思路真的很难想 主要从以下方面想：

1.要善于找到影响最大的方面 如果从最顶层开始找的话发现影响其实很小 影响最大的是在最下面那一层，会层层往上影响上面的状态 所以可以一列一列地铺 而不是一整行一整行地铺

2.既然题目中给到了n和m这两个数值，那么就应该敏感一些了 递归返回的点就在于num_a和num_b是否超过了这两个数值 如果超过了就直接return 如果到了最后那一步还是没超过的话ans++

3.要善于把这种三角形排列的问题转化为坐标问题 坐标其实很好算的

**注意事项:**

1.dfs第一步不一定是在main里面 如果第一步固定的话（路径之谜）可以把第一步放main里面 如果第一步不是固定的，也符合dfs逻辑的话，直接把第一步放到dfs里面最好

2.递归终点可能比真实下标多一个，代表着以前的下标都顺利进行到位了且没有返回 但是！！要注意传的参数，**如果在最后一步还需要检验参数正确与否的话，检验条件一定要写在递归终点前面** 不然如果不符合条件的话也会把答案情况+1
