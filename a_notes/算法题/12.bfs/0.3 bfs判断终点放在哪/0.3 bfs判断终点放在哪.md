诚如这个题，判断重点最好是放在出队的时候 理由如下：

1.不会漏掉起点状态（如果起点就是终点的话，只在入队的时候判断，那么起点永远都不可能被找到）

2.逻辑更连贯整洁，出队的时候就只处理出队的事情，顺便判断一下是否为终点，并且当前step就是答案，不用后续step+1。

```c++
#include <iostream>
#include <queue>
#include <unordered_map>
#include <string>

using namespace std;

// 结构体只存当前状态和走到这一步花的步数
struct node {
    string s;
    int step;
};

// 只有上下左右 4 个方向，不需要自己待在原地不动
int dx[4] = {0, 0, 1, -1};
int dy[4] = {1, -1, 0, 0};

void bfs(string start) {
    int pos_A = -1, pos_B = -1;
    
    // 1. 找到初始状态下 A 和 B 的一维下标
    for (int i = 0; i < 6; i++) {
        if (start[i] == 'A') pos_A = i;
        if (start[i] == 'B') pos_B = i;
    }

    queue<node> q;
    unordered_map<string, int> vis; // 绝对不能忘的查重神器

    q.push({start, 0});
    vis[start] = 1;

    while (!q.empty()) {
        string curr = q.front().s;
        int step = q.front().step;
        q.pop();

        // 2. 终点判定：原来 A 的位置变成了 B，原来 B 的位置变成了 A
        if (curr[pos_A] == 'B' && curr[pos_B] == 'A') {
            cout << step << endl;
            return;
        }

        // 3. 寻找唯一的“主角”：空格的一维下标
        int space_idx = curr.find(' ');
        
        // 核心坐标转换：一维下标转二维坐标 (x, y)
        // 对于 2行3列 的网格：行 = idx / 列数，列 = idx % 列数
        int x = space_idx / 3;
        int y = space_idx % 3;

        // 4. 让空格向四个方向移动
        for (int i = 0; i < 4; i++) {
            int nx = x + dx[i];
            int ny = y + dy[i];

            // 越界检查
            if (nx < 0 || nx >= 2 || ny < 0 || ny >= 3) continue;

            // 核心坐标转换：二维坐标 (nx, ny) 转变回一维下标
            int next_idx = nx * 3 + ny;
            
            string next_s = curr;
            // 交换空格和目标卡片的位置
            swap(next_s[space_idx], next_s[next_idx]);

            // 如果这个新排布以前没见过，就入队
            if (vis[next_s] == 0) {
                vis[next_s] = 1;
                q.push({next_s, step + 1});
            }
        }
    }
}

int main() {
    string s1, s2;
    // 应对带有空格的输入，必须用 getline
    getline(cin, s1);
    getline(cin, s2);
    
    // 将两行直接拼成一个长度为 6 的字符串
    string start = s1 + s2; 
    
    bfs(start);

    return 0;
}
```
