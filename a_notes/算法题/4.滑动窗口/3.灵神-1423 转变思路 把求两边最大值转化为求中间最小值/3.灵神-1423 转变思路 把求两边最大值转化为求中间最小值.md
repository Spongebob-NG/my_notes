```c++
class Solution {
public:
    int maxScore(vector<int>& cardPoints, int k) {
        int cnt = cardPoints.size() - k;
        long long sum = 0;
        long long ans = INT_MAX;
        int l = 0, r = 0;
        while (r<cardPoints.size()) {
            if (r - l + 1 != cnt) {
                sum += cardPoints[r++];
            }
            else {
                sum += cardPoints[r++];
                ans = min(ans, sum);
                sum -= cardPoints[l++];
            }
        }
        ans = (ans == INT_MAX) ? 0 : ans;
        long long sum1 = 0;
        for (auto x : cardPoints)sum1 += x;
        return sum1-ans;
    }
};

```

这道题可以转变思路，由于他在两边的数量是固定的，而且每次要在两边抓最大值，不如把题目转化为求中间的滑动窗口最小值
