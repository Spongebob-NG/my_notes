```c++
class Solution {
public:
    int takeCharacters(string s, int k) {
        if (k == 0)return 0;
        int n = s.length();
        unordered_map<char, int>count;
        int left = 0, right = 0;
        int ans = INT_MAX;
        for (int i = 0; i < s.length(); i++) {
            count[s[i]]++;
        }
        if (!(count['a'] >= k && count['b'] >= k && count['c'] >= k))return -1;
        while (right < n) {
            count[s[right]]--;
            while (!(count['a'] >= k && count['b'] >= k && count['c'] >= k)) {
                count[s[left]]++;
                left++;
            }
            ans = min(ans, n - (right - left + 1));
            right++;
        }
        return ans;
    }
};
```
