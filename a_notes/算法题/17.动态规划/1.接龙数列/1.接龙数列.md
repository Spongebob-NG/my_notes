这道题的思路：dp[i]->目前为止，以数字i结尾的接龙序列，最长能有多长？

```c++
#include <iostream>
#include<algorithm>
using namespace std;
const int N = 1e5 + 5;
int a[N];
int dp[15];
int n;
int main()
{
	cin >> n;
	for (int i = 0; i < n; i++) {
		string s;
		cin >> s;
		int post = s.back() - '0';
		int front = s[0] - '0';
		dp[post] = max(dp[post], dp[front] + 1);
	}
	
	int ans = 0;
	for (int i = 0; i < 10; i++)ans = max(ans, dp[i]);
	cout << n-ans << endl;

	return 0;
}
```
