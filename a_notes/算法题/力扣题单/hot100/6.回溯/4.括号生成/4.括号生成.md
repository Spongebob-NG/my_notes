这道题 自己做出来了 但是比较笨 在最后检验的时候是自己**模拟了一个栈手动检验**的 **时间复杂度太高了**  而且剪枝也没有很快 如果说**右括号的数量大于左括号数量就应该剪枝了** 空间复杂度也比较高(栈深度以及两个数组)

```c++
class Solution {
public:
    vector<string>ans;
    string s;
    char temp[2] = { '(',')' };
    int count[2];
    void helper(int n,int pos) {
        if (pos == 2 * n) {
            stack<char>my_stack;
            for (int i = 0; i < s.size(); i++) {
                if (!my_stack.empty() && s[i] == ')' && my_stack.top() == '(') {
                    my_stack.pop();
                }
                else my_stack.push(s[i]);
            }
            if (my_stack.empty()) {
                ans.push_back(s);
            }
            return;
        }
        if (count[0] > n || count[1] > n)return;
        for (int i = 0; i < 2; i++) {
            s.push_back(temp[i]);
            count[i]++;
            helper(n, pos + 1);
            s.pop_back();
            count[i]--;
        }
    }
    vector<string> generateParenthesis(int n) {
        helper(n, 0);
        return ans;
    }
};
```

之后第二个版本 最后还是用栈来检验比较的 区别就是把栈作为全局变量了 不用一直创建然后销毁 并且条件count[1]>n改成了count[1]>count[0] 不过这样还是太慢了(具体代码不贴出来了 区别不大)

下面来说最后一个效率较高的版本:**不需要用一个栈来检验逻辑** **只需要传当前的左括号以及右括号数量** 但凡**右括号数量大于左括号** 就应该**结束**了 如果说到了最后 都没有结束的话 就代表这个字符串是符合条件的

```c++
class Solution {
public:
    vector<string>ans;
    void helper(string&curr,int n, int open, int close) {
        if (curr.size() == 2 * n) {
            ans.push_back(curr);
            return;
        }
        if (open < n) {
            curr.push_back('(');
            helper(curr, n, open + 1, close);
            curr.pop_back();
        }
        if (close < open) {
            curr.push_back(')');
            helper(curr, n, open, close + 1);
            curr.pop_back();
        }
    }
    vector<string> generateParenthesis(int n) {
        string curr = "";
        helper(curr, n, 0, 0);
        return ans;
    }
};
```
