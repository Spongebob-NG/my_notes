这道题和**选集合**那道题有异曲同工之秒 现在总结出来一个经验:**凡是集合会重复的 都设定一个start** 如果说元素不能重复的话 每次递归让start+1 如果说元素可以重复的话 每次递归都从start下标开始

ps:针对这道题来说还有个需要注意的点 就是同一个数字可以**无限制重复**被选取 所以说不能让选中的数字个数到了candidates的大小就停止了 而是让**sum的和大于target时才停止**

不过这道题靠自己独立完成了 真棒.

```c++
class Solution {
public:
    int sum = 0;
    vector<vector<int>>ans;
    vector<int>temp;
    void helper(vector<int>& candidates, int target,int start) {
        if (sum == target) {
            ans.push_back(temp);
            return;
        }
        if (sum > target)return;
        for (int i = start; i < candidates.size(); i++) {
            sum += candidates[i];
            temp.push_back(candidates[i]);
            helper(candidates, target,i);
            sum -= candidates[i];
            temp.pop_back();
        }
    }
    vector<vector<int>> combinationSum(vector<int>& candidates, int target) {
        helper(candidates, target,0);
        return ans;
    }
};
```
