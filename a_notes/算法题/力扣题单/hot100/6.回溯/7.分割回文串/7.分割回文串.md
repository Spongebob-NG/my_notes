这题一开始其实没啥思路 后来提醒了一下 思路其实很明确:递归的时候参数传一个start 然后每层递归都从start开始遍历到s的末尾(因为start之前已经保证是回文串了) 所以说这道题相对来说其实不算难

下面是第一版代码:

```c++
class Solution {
public:
    vector<vector<string>>ans;
    vector<string>vec_str;
    bool is_(string& s, int start, int end) {
        int p1 = start;
        int p2 = end;
        while (p1 <= p2) {
            if (s[p1] == s[p2]) {
                p1++;
                p2--;
            }
            else return false;
        }
        return true;
    }
    void helper(string& s, int start) {
        if (start == s.length()) {
            ans.push_back(vec_str);
            return;
        }
        for (int i = start; i < s.length(); i++) {
            if (is_(s, start, i)) {
                vec_str.push_back(s.substr(start, i - start + 1));
                helper(s, i + 1);
                vec_str.pop_back();
            }
        }
    }
    vector<vector<string>> partition(string s) {
        helper(s, 0);
        return ans;
    }
};
```

这版代码在**每次**判断子串是否为回文串的时候都会进入函数来检查 **效率太慢了** 我们可以搞一个**记忆化数组** 也就是**动态规划** 来事先保存 `dp[start][end]`

这道题可以说 在某种程度上是**动态规划的启蒙** 因为要想:`dp[start][end]`是**依赖于**`dp[start+1][end-1]`的 也就是说必须要**先计算出后面的行 才能计算出前面的行** 然后 **前面的列 才能算出后面的列** 所以要想让整个dp数组都被覆盖到的话 **行就得反向遍历 列就得正向遍历**

要注意:**一定要先算完整个dp** 才能进入后续的判断 一次性就要算完 不然的话 如果在递归里面算 就会有很多个子串用不上dp 覆盖不全!!!

```c++
class Solution {
public:
    vector<vector<string>>ans;
    vector<string>vec_str;
    vector<vector<bool>>dp;
    void helper(string& s, int start) {
        if (start == s.length()) {
            ans.push_back(vec_str);
            return;
        }
        for (int i = start; i < s.length(); i++) {
            if (dp[start][i]) {
                vec_str.push_back(s.substr(start, i - start + 1));
                helper(s, i + 1);
                vec_str.pop_back();
            }
        }
    }
    vector<vector<string>> partition(string s) {
        dp.assign(s.length(), vector<bool>(s.length(), true));
        for (int i = s.length() - 1; i >= 0; i--) {
            for (int j = i + 1; j < s.length(); j++) {
                dp[i][j] = (s[i] == s[j]) && dp[i + 1][j - 1];
            }
        }
        helper(s, 0);
        return ans;
    }
};
```
