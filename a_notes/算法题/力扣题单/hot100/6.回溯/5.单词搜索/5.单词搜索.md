这道题 最开始的代码是从0,0开始遍历 这样肯定不对 因为**这个是dfs 无论后面选还是不选 开头都是定死的 s都是从0,0开始递归增加** 所以肯定不对 因为**每个格子都能作为开头**. 所以接下来第一个版本的代码就是 把从0,0开头改成 从字符跟word[0]相同的位置开始递归.

```c++
class Solution {
public:
    string s;
    vector<int>dx = { 0,0,1,-1 };
    vector<int>dy = { 1,-1,0,0 };
    bool find = false;
    void helper(vector<vector<char>>& board, string& word,int x,int y,vector<vector<bool>>&flag) {
        if (find)return;
        for (int i = 0; i < 4; i++) {
            int xi = x + dx[i];
            int yi = y + dy[i];
            if (xi < 0 || xi >= board.size() || yi < 0 || yi >= board[0].size() || flag[xi][yi])continue;
            flag[xi][yi] = true;
            s.push_back(board[xi][yi]);

            if (s == word) {
                find = true;
                return;
            }
            helper(board, word, xi, yi, flag);
            flag[xi][yi] = false;
            s.pop_back();
        }
    }
    bool exist(vector<vector<char>>& board, string word) {
        vector<vector<bool>>flag(board.size(), vector<bool>(board[0].size()));
        for (int i = 0; i < board.size(); i++) {
            for (int j = 0; j < board[0].size(); j++) {
                if (board[i][j] == word[0]) {
                    flag[i][j] = true;
                    s.push_back(board[i][j]);
                    if (s == word) {
                        return true;
                    }
                    helper(board, word, i, j, flag);
                    flag[i][j] = false;
                    s.pop_back();
                }
            }
        }
        return find;
    }
};
```

而这个版本的代码 只能过一部分样例 为啥呢 **时间复杂度太高了**.我们需要剪枝

如果说 **但凡当前字符跟word的这个字符不匹配了 就return** 这样就能做到有效剪枝 时间复杂度大大降低 (并且别忘了这道题 如果**word长度为1 并且直接匹配到了的话 就直接返回true**了 如果不返回的话 后面进入递归之后 一直都返回不了true的 会被忽略)

```c++
class Solution {
public:
    string s;
    int dx[4] = { 0,0,1,-1 };
    int dy[4] = { 1,-1,0,0 };
    bool find = false;
    void helper(vector<vector<char>>& board, string& word, int x,int y,int cur_index,vector<vector<bool>>&flag) {
        if (find||cur_index >= word.length())return;
        for (int i = 0; i < 4; i++) {
            int xi = x + dx[i];
            int yi = y + dy[i];
            if (xi < 0 || yi < 0 || xi >= board.size() || yi >= board[0].size() || flag[xi][yi] || board[xi][yi] != word[cur_index])continue;
            s.push_back(board[xi][yi]);
            if (s == word) {
                find = true;
                return;
            }
            flag[xi][yi] = true;
            helper(board, word, xi, yi, cur_index + 1, flag);
            s.pop_back();
            flag[xi][yi] = false;
        }
    }
    bool exist(vector<vector<char>>& board, string word) {
        vector<vector<bool>>flag(board.size(), vector<bool>(board[0].size()));
        for (int i = 0; i < board.size(); i++) {
            for (int j = 0; j < board[0].size(); j++) {
                if (board[i][j] == word[0]) {
                    flag[i][j] = true;
                    s.push_back(word[0]);
                    if (s == word)return true;
                    helper(board, word, i, j, 1,flag);
                    s.pop_back();
                    flag[i][j] = false;
                }
            }
        }
        return find;
    }
};
```
