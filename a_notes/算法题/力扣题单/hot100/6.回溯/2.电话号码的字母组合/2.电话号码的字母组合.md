这道题 两个示例都过了 所以没注意后面一个数字可能对应4个字母的情况:

```c++
class Solution {
public:
    vector<string>ans;
    string s;
    void helper(string& digits, int n, vector<bool>& flag) {
        if (n == digits.size()) {
            ans.push_back(s);
            return;
        }
        for (char i = 'a' + (digits[n] - '2') * 3; i < 'a' + (digits[n] - '1') * 3; i++) {
            if (flag[i - 'a'])continue;
            flag[i - 'a'] = true;
            s.push_back(i);
            helper(digits, n + 1, flag);
            flag[i - 'a'] = false;
            s.pop_back();
        }
    }
    vector<string> letterCombinations(string digits) {
        vector<bool>flag(26,false);
        helper(digits, 0, flag);
        return ans;
    }
};
```

后面一个个手动分所有情况设定start和end的值之后:

```c++
class Solution {
public:
    vector<string>ans;
    string s;
    void helper(string& digits, int n, vector<bool>& flag) {
        if (n == digits.size()) {
            ans.push_back(s);
            return;
        }
        char start = ' ';
        char end = ' ';
        if (digits[n] <= '6') {
            start='a' + (digits[n] - '2') * 3;
            end = start + 3;
        }
        else if (digits[n] == '7') {
            start = 'a' + 15;
            end = start + 4;
        }
        else if (digits[n] == '8') {
            start = 'a' + 19;
            end = start + 3;
        }
        else {
            start = 'a' + 22;
            end = start + 4;
        }
        for (char i =start ; i < end; i++) {
            if (flag[i - 'a'])continue;
            flag[i - 'a'] = true;

```

但是最终还是过不了所有的样例 因为字符串里面的字母可以一样 不应该有flag这个条件 并且之前冗余的if-else可以用一个类似于哈希表功能的数组来替换 这样能显著提高时间复杂度

最终答案:

```c++
class Solution {
public:
    vector<string>ans;
    string s;
    vector<string>mapping = {
        "abc",
        "def",
        "ghi",
        "jkl",
        "mno",
        "pqrs",
        "tuv",
        "wxyz"
    };
    void helper(string& digits, int n) {
        if (n == digits.size()) {
            ans.push_back(s);
            return;
        }
        string temp = mapping[digits[n] - '2'];
        int length = temp.size();
        for (int i = 0; i < length; i++) {
            s.push_back(temp[i]);
            helper(digits, n + 1);
            s.pop_back();
        }
    }
    vector<string> letterCombinations(string digits) {
        vector<bool>flag(26,false);
        helper(digits, 0);
        return ans;
    }
};
```
