这道题 核心其实就是弄三个数组 一个col 一个diagonal 一个anti_diagonal(后两个是主对角线和副对角线的意思) 为啥不需要row呢 因为row直接在for循环里面递归 参数填row+1就行了 不会占到同一行row的  要注意 主对角线和副对角线开的大小要大一点 因为如果是主对角线的话 col和row的差是恒定的 并且可能会出现负数 所以要加上n.副对角线相加也可能超出n 所以最好大小都开两倍

```c++
class Solution {
public:
    bool col[10];
    bool diagonal[20];
    bool anti_diagonal[20];
    vector<vector<string>>ans;
    vector<string>vec_str;
    string s;
    void helper(vector<vector<char>>&map,int& n, int cur_row) {
        if (cur_row == n) {
            for (auto& x : map) {
                for (auto& y : x) {
                    s.push_back(y);
                }
                vec_str.push_back(s);
                s = "";
            }
            ans.push_back(vec_str);
            vec_str = vector<string>();
            return;
        }
        for (int i = 0; i < n; i++) {
            if (col[i] || diagonal[cur_row - i + n] || anti_diagonal[cur_row + i])continue;
            map[cur_row][i] = 'Q';
            col[i] = true;
            diagonal[cur_row - i + n] = true;
            anti_diagonal[cur_row + i] = true;
            helper(map, n, cur_row + 1);
            col[i] = false;
            diagonal[cur_row - i + n] = false;
            anti_diagonal[cur_row + i] = false;
            map[cur_row][i] = '.';
       }
    }
    vector<vector<string>> solveNQueens(int n) {
        vector<vector<char>>map(n, vector<char>(n,'.'));
        helper(map, n, 0);
        return ans;
    }
};
```
