最开始的错误解法（能通过部分样例）

```c++
  using node = ListNode*;
class Solution {
public:
    long long calList(node list,long long num) {
        if (!list)return 0;
        return list->val*num + calList(list->next,num*10);
    }
    node nodeCreate(long long num) {
        if (num == 0)return nullptr;
        int digit = num % 10;
        num /= 10;
        node p = new ListNode(digit);
        p->next = nodeCreate(num);
        return p;
    }
    ListNode* addTwoNumbers(ListNode* l1, ListNode* l2) {
        long long sum = calList(l1,1) + calList(l2,1);
        if (sum == 0)return new ListNode(0);
        return nodeCreate(sum);
    }
};
```

这个解法  一个是递归调用太消耗性能了  再一个就是位数太大了 用long long都存不下 只能用**高精度加法**

所以对于这种**长链表**  最好的办法只有**循环** 不能递归



正确解法：

```c++
  using node = ListNode*;
  class Solution {
  public:
      ListNode* addTwoNumbers(ListNode* l1, ListNode* l2) {
          ListNode dummy(0);
          node tail = &dummy;
          int carry = 0;
          while (l1 || l2 || carry) {
              int sum = carry;
              if (l1) {
                  sum += l1->val;
                  l1 = l1->next;
              }
              if (l2) {
                  sum += l2->val;
                  l2 = l2->next;
              }
              tail->next = new ListNode(sum % 10);
              tail = tail->next;
              carry = sum / 10;
              
          }
          return dummy.next;
      }
  };
```

不过由于每次都会new个新节点出来 开销也挺大的  所以可以直接**复用l1** 也就是说**tail->next可以直接指向l1**  只有**当l1为空链表的时候才需要new新节点**  这里就不再过多赘述
