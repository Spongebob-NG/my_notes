```c++
using node = ListNode*;
class Solution {
public:
    ListNode* mergeTwoLists(ListNode* list1, ListNode* list2) {
        if (!list1)return list2;
        else if (!list2) return list1;
        node p = list1, q = list2;
        node ans = nullptr;
        node now = nullptr;
        if (p->val <= q->val) {
            ans = p;
            now = ans;
            p = p->next;
        }
        else {
            ans = q;
            now = ans;
            q = q->next;
        }
        while (p && q) {
            if (p->val <= q->val) {
                now->next = p;
                now = p;
                p = p->next;
            }
            else {
                now->next = q;
                now = q;
                q = q->next;
            }
        }
        while (p) {
            now->next = p;
            now = p;
            p = p->next;
        }
        while (q) {
            now->next = q;
            now = q;
            q = q->next;
        }

        return ans;
    }
};
```

注意  一开始一直超出时间限制 **形成了循环链表** 是因为在**初始化**ans的时候 **p或者q没有动**  应该让p或者q再**往后移一个位置**的 不然会死循环
