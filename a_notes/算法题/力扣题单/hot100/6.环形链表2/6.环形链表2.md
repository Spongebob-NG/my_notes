这道题也是需要记住一个比较巧妙的 通过数学公式推导出的一个结论 就是 先搞一对快慢指针 如果相遇了的话  再搞个指针从head开始 然后再跟slow相遇 就是入环点  这个先记住就好了

```c++
  using node = ListNode*;
  class Solution {
  public:
      ListNode* detectCycle(ListNode* head) {
          if (!head||!head->next)return nullptr;
          node fast = head, slow = head;
          bool flag = false;
          while (fast&&fast->next) {
              slow = slow->next;
              fast = fast->next->next;
              if (fast == slow) {
                  flag = true;
                  break;
              }
          }
          if (flag) {
              node temp = head;
              while (temp != slow) {
                  temp = temp->next;
                  slow = slow->next;
              }
              return slow;
          }
          else return nullptr;

      }
  };
```
