这道题一开始手动建栈 虽然最终通过AI的修正 至少能够过样例 但是还是太麻烦了

所以这里有必要说一下：**能用显式栈解决的问题 都能用递归来做 而且逻辑更清晰** 反之亦然

```c++
class Solution {
public:
    string decodeString(string s) {
        if (s.empty())return s;
        for (auto& c : s) {
            if (isalpha(c)) {
                return c + decodeString(s.substr(1));
            }
            else {
                int i = s.find('[');
                int cnt = 1;
                for (int j = i + 1; j < s.length(); j++) {
                    if (s[j] == '[')cnt++;
                    else if (s[j] == ']') {
                        cnt--;
                        if (cnt == 0) {
                            int num = stoi(s.substr(0, i));
                            string temp = decodeString(s.substr(i + 1, j - 1 - i));
                            string res = "";
                            for (int i = 0; i < num; i++) {
                                res += temp;
                            }
                            return res + decodeString(s.substr(j + 1));
                        }
                    }
                }
            }
        }
        return "";
    }
};
```
