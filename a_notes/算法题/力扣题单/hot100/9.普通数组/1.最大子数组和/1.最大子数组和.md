这道题做过很多遍 但每次遇到其实都有点懵 一看到子数组求和就想着要前缀和+哈希 但是**这道题不是找某个固定sum的题  而是求最大值** 所以**第一种**方案就是要用到**动态规划**的思想

```c++
class Solution {
public:
    int maxSubArray(vector<int>& nums) {
        int pre = 0, ans = INT_MIN;
        for (auto& x : nums) {
            pre = max(pre + x, x);
            ans = max(ans, pre);
        }
        return ans;
    }
};
```

第二种方案 针对这道题来说 可以用**分治法**  因为求最大子数组和的时候 和最大的子数组 **可以在左边 可以在右边 也可以经过中间** 所以每个**大问题能分解成小问题** 最后再合并

```c++
class Solution {
public:
    int helper(vector<int>& nums, int left, int right) {
        int mid = (right - left) / 2 + left;
        int sum_left = INT_MIN, sum_right = INT_MIN;
        int sum = 0;
        for (int i = mid; i >= left; i -- ) {
            sum += nums[i];
            sum_left = max(sum_left, sum);
        }
        sum = 0;
        for (int i = mid + 1; i <= right; i++) {
            sum += nums[i];
            sum_right = max(sum_right, sum);
        }
        return sum_left + sum_right;
    }
    int sub_helper(vector<int>& nums, int left,int right) {
        if (left >= right)return nums[left];
        int mid = (right - left) / 2 + left;
        int left_max = sub_helper(nums, left, mid);
        int right_max = sub_helper(nums, mid + 1, right);
        int mid_max = helper(nums, left, right);
        return max(max(left_max, right_max), mid_max);
    }
    int maxSubArray(vector<int>& nums) {
        return sub_helper(nums, 0, nums.size() - 1);
    }
};
```
