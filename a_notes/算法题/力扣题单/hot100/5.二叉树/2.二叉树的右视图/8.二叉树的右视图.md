这道题一开始思路想出来了 就是说层序遍历 然后到了每层最后一个节点的时候再放到ans数组里面.但是一开始想的是再定义一个额外的结构体来存储每个节点的高度 但是太麻烦了而且空间复杂度高 并且实在想不出别的办法了. 这里可以说明一下 **层序遍历是可以完全遍历完当前这一层之后再遍历下一层的** 只需要每层都弄个size记录当前层的队列的size()大小 然后for循环遍历size次就行了

```c++
 using node = TreeNode*;
 class Solution {
 public:
     vector<int> rightSideView(TreeNode* root) {
         if (!root)return {};
         vector<int>ans;
         queue<node>q;
         q.push(root);
         while (!q.empty()) {
             int size = q.size();
             for (int i = 0; i < size; i++) {
                 if (i == size - 1) {
                     ans.push_back(q.front()->val);
                  }
                 if (q.front()->left) { q.push(q.front()->left); }
                 if (q.front()->right) { q.push(q.front()->right); }
                 q.pop();
             }
         }
         return ans;
     }
 };
```

然后还有个更方便的办法 就是 根->右->左 这样遍历 然后参数里面加个int h来记录当前的高度就行了

```c++
using node = TreeNode*;
class Solution {
public:
    void helper(node root, vector<int>& ans, int h) {
        if (!root)return;
        if (h > ans.size()) {
            ans.push_back(root->val);
        }
        helper(root->right, ans, h + 1);
        helper(root->left, ans, h + 1);
    }
    vector<int> rightSideView(TreeNode* root) {
        vector<int>ans;
        helper(root, ans, 1);
        return ans;
    }
};

```
