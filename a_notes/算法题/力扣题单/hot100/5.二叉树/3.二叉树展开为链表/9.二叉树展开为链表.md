这道题一直想着说要弄个新的链表 但是题目的要求是要在原地改 所以代码如下

要注意的几个点:

1.tail传参的时候一定要引用传参 不然改变不了指针的指向 这个坑踩过很多次了

2.root->left 和root->right要事先存起来 因为后面会改这两个的指向

```c++
using node = TreeNode*;
class Solution {
public:
    void pre_order(node root,node&tail) {
        if (!root)return;
        node temp_l = root->left;
        node temp_r = root->right;
        if (!tail)tail = root;
        else {
            tail->right = root;
            tail->left = nullptr;
            tail = root;
        }
        pre_order(temp_l, tail);
        pre_order(temp_r, tail);
    }
    void flatten(TreeNode* root) {
        node tail = nullptr;
        pre_order(root, tail);
    }
};
```
