```c++
class Solution {
public:
    vector<int> maxSlidingWindow(vector<int>& nums, int k) {
        priority_queue<int, vector<int>, less<int>>q;
        unordered_map<int, int>count;
        vector<int>ans;
        int left = 0, right = 0;
        int n = nums.size();
        while (right < n) {
            if (right < k - 1) {
                count[nums[right]]++;
                q.push(nums[right]);
                right++;
                continue;
            }
            count[nums[right]]++;
            q.push(nums[right]);
            right++;
            ans.push_back(q.top());
            count[nums[left]]--;
            while (!q.empty()&&count[q.top()] == 0) {
                q.pop();
            }
            left++;
        }
        return ans;
    }
};
```

这是最开始想到的 优先队列+哈希表  虽然中间在 while (!q.empty()&&count[q.top()] == 0) 这个条件上面的判断有点失误(之前还判断了一下num[left]是否等于q.top() 但其实没必要 只要count等于0了就应该删掉) 但是逻辑还是能想得到.只不过**哈希表的效率太低了**

所以这里直接用到pair<int,int>来存储 如果下标不在窗口之内 直接删掉就行了 免去了哈希表的大开销

```c++
using PII = pair<int, int>;
class Solution {
public:
    vector<int> maxSlidingWindow(vector<int>& nums, int k) {
        priority_queue<PII, vector<PII>, less<PII>>q;
        vector<int>ans;
        int left = 0, right = 0;
        int n = nums.size();
        while (right < n) {
            if (right < k - 1) {
                q.push({ nums[right],right });
                right++;
                continue;
            }
            q.push({ nums[right],right });
            right++;
            ans.push_back(q.top().first);
            left++;
            while (!q.empty() && q.top().second < left) {
                q.pop();
            }
        }
        return ans;
    }
};
```

但其实这道题最好的方法是用**单调队列**  如果说太老的话就直接出队  如果队末的元素比新来的元素要菜的话 就卷铺盖走人 直到让新人入队

```c++
class Solution {
public:
    vector<int> maxSlidingWindow(vector<int>& nums, int k) {
        vector<int>ans;
        deque<int>q;
        for (int i = 0; i < nums.size(); i++) {
            if (!q.empty() && q.front() <= i - k) {
                q.pop_front();
            }
            while (!q.empty() && nums[q.back()] < nums[i]) {
                q.pop_back();
            }
            q.push_back(i);
            if (i >= k-1) {
                ans.push_back(nums[q.front()]);
            }
        }
        return ans;
    }
};
```
