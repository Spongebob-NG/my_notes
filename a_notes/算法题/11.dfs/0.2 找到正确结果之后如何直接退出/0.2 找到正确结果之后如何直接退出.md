```c++
#include <iostream>
#include<vector>
using namespace std;
int dx[4] = { 0,0,1,-1 };
int dy[4] = { 1,-1,0,0 };
int l1[25], l2[25];
int m[25][25];
int f[25][25];
int n;
vector<int>ans;
void dfs(int i, int j, int sum) {
	if (i==n-1&&j==n-1) {
		if (sum == 0) {
			for (int i = 0; i < ans.size(); i++) {
				cout << ans[i] << ' ';
			}
			exit(0);
		}
		return;
	}
	
	for (int m = 0; m < 4; m++) {
		int xi = i + dx[m];
		int yi = j + dy[m];
		if (xi < 0 || xi >= n || yi < 0 || yi >= n||f[xi][yi]==1)continue;
		if (l1[xi] && l2[yi]) {
			ans.push_back(xi * n + yi);
			f[xi][yi] = 1;
			l1[xi]--; l2[yi]--;
			sum-=2;
			dfs(xi, yi, sum);
			f[xi][yi] = 0;
			l1[xi]++; l2[yi]++;
			sum+=2;
			ans.pop_back();
		}

	}
}
int main()
{
	cin >> n;
	for (int i = 0; i < n; i++)cin >> l2[i];
	for (int i = 0; i < n; i++)cin >> l1[i];
	int sum = 0;
	for (int i = 0; i < n; i++) {
		sum += l1[i];
		sum += l2[i];
	}
	ans.push_back(0);
	f[0][0] = 1;
	l1[0]--;
	l2[0]--;
	sum -= 2;
	dfs(0,0,sum);


	return 0;
}
```

从这道题可以悟出针对这种**dfs的套路**以及找到正确结果之后**结束递归的方法**：

------



你纠结的这两个点，可以说是所有初学者在刷 DFS 时必定会撞上的两堵墙。你能自己意识到“`return` 会退回到上一层继续搜”，说明你已经在脑海里建立起了一棵非常清晰的递归搜索树了！你的直觉非常敏锐且完全正确。

今天咱们就彻底把这两个纠结点砸碎，给你总结一套**“蓝桥杯迷宫/网格 DFS 万能心法”**。

------

### 纠结点一：起点 (0,0) 的操作，写在 `main` 里还是 `dfs` 里？

**结论先行：强烈建议写在 `main` 里。**

你可以用**“下棋”**来类比：

- **`main` 函数**：负责**“把棋子摆到起点上”**。所以你需要把起点的坐标加入路径、标记起点已访问、扣除起点消耗的资源。
- **`dfs` 函数**：负责**“站在当前格子上，往四周看，并迈出下一步”**。

**如果强行写在 `dfs` 里会怎样？**

你会写得极其难受。因为 `dfs` 函数每一层都会执行一遍里面的代码，如果你把“起点加入路径、扣除资源”写在 `dfs` 开头，你就必须加一个恶心的特判：

```
if (x == 0 && y == 0) { ... }
```

而且这种写法极容易导致逻辑错乱，比如每次递归回来撤销标记时，不小心把起点的标记也给撤销了。

**黄金法则**：在调用 `dfs(起点)` 之前，**先把起点的所有“后事”在 `main` 里料理干净**。进入 `dfs` 后，里面只有纯粹的“向四周探索”的逻辑。

------

### 纠结点二：找到终点后，怎么“全身而退”？

你的担忧是 100% 正确的：**在深层递归中，一个单纯的 `return;` 只能结束当前这一层。** 它会退回到上一层的 `for` 循环里，然后你的程序会顺着另一个方向继续往下搜。

这就是为什么我教你用 `exit(0);`（直接拔电源，强行终止整个程序）。对于“只需要找**一条**合法路径”的题目，`exit(0)` 是竞赛中最爽的黑科技。

**但如果题目要求找“最短路径”，或者不允许用 `exit(0)`，怎么用 `return` 优雅地退出来？**

标准做法是：**引入一个全局的“信号枪”（`bool` 变量）。**

我们来看看代码怎么写：

C++

```
bool found = false; // 信号枪，初始为未找到

void dfs(int x, int y) {
    // 1. 如果信号枪已经打响了，啥也别干，立刻撤退！(核心连锁反应)
    if (found) return; 

    // 2. 找到终点了！
    if (x == n - 1 && y == n - 1) {
        if (check()) {
            print_ans();
            found = true; // 打响信号枪！
            return;       // 退回上一层
        }
        return;
    }

    // 3. 四个方向探索
    for (int k = 0; k < 4; k++) {
        int nx = x + dx[k];
        int ny = y + dy[k];

        if (越界 || 已访问 || 不满足条件) continue;

        // 做出选择
        f[nx][ny] = 1;
        ans.push_back(...);
        
        dfs(nx, ny); // 往下搜

        // 【关键点】：搜完回来后，先看一眼信号枪！
        // 如果底下的某一层已经找到了答案把 found 变成了 true，
        // 那我这一层也绝对不能再回溯（撤销标记）或者继续循环了，直接 return 逃跑！
        if (found) return; 

        // 撤销选择 (只有没找到答案，才会执行到这里，继续尝试其他路)
        f[nx][ny] = 0;
        ans.pop_back();
    }
}
```

**这个 `bool found` 就像多米诺骨牌：**

一旦最深处触发了 `found = true`，它上一层紧跟着的 `if (found) return;` 就会被触发，然后上上一层也被触发……瞬间，成百上千层的调用栈就像雪崩一样全部 `return` 清空，干脆利落。

### 🏆 蓝桥杯 DFS 网格题万能模板

把你这两天踩的坑、悟出的道理，浓缩成下面这个可以直接默写的模板：

C++

```
#include <iostream>
#include <vector>
using namespace std;

// 1. 方向数组（死背下来）
int dx[4] = {-1, 1, 0, 0};
int dy[4] = {0, 0, -1, 1};

int vis[50][50]; // 访问标记
bool found = false; // 连锁退出的信号枪

void dfs(int x, int y) {
    // 防御性拦截（针对连锁退出）
    if (found) return;

    // 1. 终点结算
    if (x == 终点x && y == 终点y) {
        if (满足额外限制) {
            输出答案;
            found = true; // 触发多米诺骨牌
            return;       // 或用 exit(0); 暴力结束
        }
        return;
    }

    // 2. 核心扩散逻辑
    for (int k = 0; k < 4; k++) {
        int nx = x + dx[k];
        int ny = y + dy[k];

        // 3. 越界与合法性检查 (全写在 continue 里，保持代码扁平)
        if (nx < 0 || nx >= n || ny < 0 || ny >= m) continue;
        if (vis[nx][ny] == 1) continue;
        if (不满足其他题目条件) continue;

        // 4. 标准回溯三部曲
        vis[nx][ny] = 1;      // 踩下去
        维护其他状态(如靶子--);
        
        dfs(nx, ny);          // 递归
        
        if (found) return;    // 连锁退出拦截（关键！）
        
        vis[nx][ny] = 0;      // 抬起脚
        维护其他状态(如靶子++);
    }
}

int main() {
    // 读入数据...
    
    // 5. 起点初始化大礼包 (全在 main 里做)
    vis[起点x][起点y] = 1;
    维护起点带来的其他状态变化;
    
    dfs(起点x, 起点y);
    
    return 0;
}
```
