```c++
#include<iostream>
using namespace std;
int pos = 0;
string s;
int dfs() {
	int res = 0;
	while (pos < s.length()) {
		if (s[pos] == '(') {
			pos++;
			res += dfs();
		}
		else if (s[pos] == ')'){
			pos++;
			return res;
		}
		else if (s[pos] == '|') {
			pos++;
			return max(res, dfs());
		}
		else if (s[pos] == 'x') {
			res++;
			pos++;
		}
	}
	return res;
}
int main() {
	cin >> s;
	cout << dfs() << endl;
}
```

这个dfs有点特殊 因为不是那种走迷宫一样每种情况尝试一遍，而是说去“剥洋葱”。更形象地说，遇到这种解析的题，计算机能利用函数的递归调用栈，硬生生把线性的字符串，压成一棵有层级的树。

而基本骨架就是长下面那样（拿这道题举例子来说）

```c++
// 通用解析模板
int dfs(int& pos, string& s) {
    int res = 0; // 记录当前层级的结果
    
    // 1. 横向推进：只要没越界，就在当前层级不断往后读
    while (pos < s.length()) {
        
        // 2. 遇到进入嵌套的标志（如左括号） -> 递归深搜
        if (s[pos] == '(') {
            pos++; // 消耗掉 '('
            res += dfs(pos, s); // 累加子层级的结果
        }
        
        // 3. 遇到退出嵌套的标志（如右括号） -> 返回结果，结束当前层
        else if (s[pos] == ')') {
            pos++; // 消耗掉 ')'
            return res; // ！！！强制下班！！！
        }
        
        // 4. 遇到逻辑分支/优先级标志（如 '|' 或 '+', '*'） -> 结算并进入下一分支
        else if (s[pos] == '|') {
            pos++; // 消耗掉 '|'
            // '|' 的特殊性在于它将当前层级切成了两半，所以比较完就要强制下班
            return max(res, dfs(pos, s)); 
        }
        
        // 5. 遇到基础原子（如 'x' 或 数字） -> 直接处理
        else if (s[pos] == 'x') {
            res++;
            pos++; 
        }
    }
    
    // 走到字符串尽头，返回最终结果
    return res;
}
```
