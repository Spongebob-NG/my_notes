```c++
#include <iostream>
using namespace std;
const int N = 55;
char g[N][N];
int st[N][N];
int dx[] = {0,0,-1,1,-1,-1,1,1};
int dy[] = { -1,1,0,0,-1,1,-1,1 };
int n, m;
int cnt = 0;
void dfs(int x, int y) {
	st[x][y] = 1;
	for (int i = 0; i < 4; i++) {
		int xi = x + dx[i];
		int yi = y + dy[i];
		if (xi < 0 || xi >= n+2 || yi < 0 || yi >= m+2||st[xi][yi])continue;
		if(g[xi][yi]=='1')dfs(xi, yi);
	}
}
void dfs1(int x, int y) {
	st[x][y] = 1;
	for (int i = 0; i < 8; i++) {
			int xi = x + dx[i];
			int yi = y + dy[i];
			if (xi < 0 || xi >= n+2 || yi < 0 || yi >= m+2||st[xi][yi])continue;
			if (g[xi][yi] == '1') {
				cnt++;
				dfs(xi, yi);
			}
			else dfs1(xi, yi);
	}
}
int main()
{
	int T; cin >> T;
	while (T--) {
		memset(st, 0, sizeof(st));
		memset(g, ' ', sizeof(g));
		cnt = 0;
		cin >> n >> m;
		for (int i = 1; i <= n; i++) {
			for (int j = 1; j <= m; j++) {
				cin >> g[i][j];
			}
		}
		dfs1(0, 0);
		cout << cnt << endl;
	}


	return 0;
}
```

这道题 思维是“正难则反”。如果正着统计岛屿个数，发现非常复杂且无从下手，并且内部的岛屿是不会被计算的，此时就应该反着思考：我应该是水

此题要注意以下几点：

1.水的蔓延方向有8个

2.水也要跟陆地一样，遍历的时候就要标记，不然的话会无限递归从而爆栈

3.水是作为最外圈的，所以第0行和第n+1行都是水，1到n行才是陆地真正在的地方。所以边界条件有变，不能是大于n或者m就continue了，边界要扩大点



其实这道题还能用BFS来做：

```c++
typedef pair<int, int> PII;

void bfs_land(int sx, int sy) {
    queue<PII> q;
    q.push({sx, sy});
    st[sx][sy] = true;
    while(!q.empty()) {
        PII t = q.front(); q.pop();
        for(int i = 0; i < 4; i++) { // 陆地 4 连通
            int nx = t.first + dx[i], ny = t.second + dy[i];
            if(nx >= 0 && nx < n + 2 && ny >= 0 && ny < m + 2 && g[nx][ny] == 1 && !st[nx][ny]) {
                st[nx][ny] = true;
                q.push({nx, ny});
            }
        }
    }
}

void bfs_sea(int sx, int sy) {
    queue<PII> q;
    q.push({sx, sy});
    st[sx][sy] = true;
    while(!q.empty()) {
        PII t = q.front(); q.pop();
        for(int i = 0; i < 8; i++) { // 海水 8 连通
            int nx = t.first + dx[i], ny = t.second + dy[i];
            if(nx < 0 || nx >= n + 2 || ny < 0 || ny >= m + 2 || st[nx][ny]) continue;
            if(g[nx][ny] == 1) {
                cnt++;
                bfs_land(nx, ny); // 发现岛屿，用陆地 BFS 涂黑
            } else {
                st[nx][ny] = true;
                q.push({nx, ny}); // 是海水，继续扩散
            }
        }
    }
}
```
