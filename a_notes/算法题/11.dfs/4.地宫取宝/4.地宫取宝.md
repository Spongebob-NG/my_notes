这道题其实一开始能想到正确的方向，就是dfs的四个参数，并且设置了一个全局变量ans。然而这样的方法时间复杂度实在是太高了，所以得引入记忆化数组。能用记忆化数组的原因：（动态规划）

<img src="4.%E5%9C%B0%E5%AE%AB%E5%8F%96%E5%AE%9D.assets/image-20260304214130449.png" alt="image-20260304214130449" style="zoom:50%;" />

```c++
#include<iostream>
using namespace std;
int n, m, k;
int g[55][55];
int f[55][55][15][15];
int dx[2] = { 0,1 };
int dy[2] = { 1,0 };
#define mod 1000000007
long long dfs(int x, int y, int cur_get, int cur_max) {
	if (f[x][y][cur_get][cur_max + 1]!=-1)return f[x][y][cur_get][cur_max + 1];
	if (x == n - 1 && y == m - 1) {
		if (cur_get == k) {
			return 1;
		}
		return 0;
	}
	long long dfs_ways = 0;
	for (int i = 0; i < 2; i++) {
		int xi = x + dx[i];
		int yi = y + dy[i];
		if (xi < 0 || xi >= n || yi < 0 || yi >= m)continue;
		if (g[xi][yi] > cur_max) {
			dfs_ways = (dfs_ways + dfs(xi, yi, cur_get + 1, g[xi][yi])) % mod;
			dfs_ways = (dfs_ways + dfs(xi, yi, cur_get, cur_max)) % mod;

		}
		else {
			dfs_ways = (dfs_ways + dfs(xi, yi, cur_get, cur_max)) % mod;
		}
	}
	f[x][y][cur_get][cur_max + 1] = dfs_ways;
	return dfs_ways;
	
}
int main() {
	cin >> n >> m >> k;
	for (int i = 0; i < n; i++) {
		for (int j = 0; j < m; j++) {
			cin >> g[i][j];
		}
	}
	memset(f, -1, sizeof(f));
	long long ans = 0;
	ans = (ans + dfs(0, 0, 0, -1)) % mod;
	ans = (ans + dfs(0, 0, 1, g[0][0])) % mod;
	cout << ans << endl;
}
```
