**错误版本：**

```c++
 class Solution {
 public:
     //flag:父亲是否为偶数
     int helper(TreeNode* root,bool flag, int sum) {
         if (!root)return 0;
         if (flag == 1) {
             if (root->left)sum += root->left->val;
             if (root->right)sum += root->right->val;
         }
         if (root->val % 2 == 0)return sum+helper(root->left, 1, sum) + helper(root->right, 1, sum);
         else return sum+helper(root->left, 0, sum) + helper(root->right, 0, sum);
     }
     int sumEvenGrandparent(TreeNode* root) {
         return helper(root, 0, 0);
     }
 };
```

这个逻辑是错误的 如果是想一层一层往上传的话，应该只能算从自己开始的sum值，加上自己的子节点的sum和，才是这一层干的事情，然后返回到上一层去。如果继承了上面的sum的话，绝对会**重复加**，就算这一层一无所获，也会返回上面的sum的值

**正确的做法：**

```c++
class Solution {
public:
    //flag:父亲是否为偶数
    int helper(TreeNode* root,bool flag) {
        if (!root)return 0;
        int sum = 0;
        if (flag == 1) {
            if (root->left)sum += root->left->val;
            if (root->right)sum += root->right->val;
        }
        if (root->val % 2 == 0)return sum+helper(root->left, 1) + helper(root->right, 1);
        else return sum+helper(root->left, 0) + helper(root->right, 0);
    }
    int sumEvenGrandparent(TreeNode* root) {
        return helper(root, 0);
    }
};
```

**或者说直接保险一点，只引用传参：**

```c++
class Solution {
public:
    //flag:父亲是否为偶数
    void helper(TreeNode* root,bool flag, int& sum) {
        if (!root)return ;
        if (flag == 1) {
            if (root->left!=NULL)sum += root->left->val;
            if (root->right!=NULL)sum += root->right->val;
        }
        if (root->val % 2 == 0) {
            helper(root->left, 1, sum);
            helper(root->right, 1, sum);
        }
        else {
            helper(root->left, 0, sum);
            helper(root->right, 0, sum);
        }
    }
    int sumEvenGrandparent(TreeNode* root) {
        int sum = 0;
        helper(root, 0, sum);
        return sum;
    }
};
```
