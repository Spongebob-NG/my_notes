```c++
#include <iostream>
#include<unordered_set>//防止交叉
#include<queue>
using namespace std;
int dx[8] = { -1,-1,0,1,1,1,0,-1 };
int dy[8] = { 0,1,1,1,0,-1,-1,-1 };
int number[8] = { 0,1,2,3,4,5,6,7 };
const int N = 15;
int g[N][N];
int st[N][N];
int left_to_right[N][N];
int right_to_left[N][N];
int n,k;
string s;
void dfs(int x, int y, int cur_cnt) {
	if (x == n - 1 && y == n - 1 && cur_cnt == n * n) {
		cout << s << endl;
		exit(0);
	}
	
	st[x][y] = 1;
	for (int i = 0; i < 8; i++) {
		int xi = x + dx[i];
		int yi = y + dy[i];
		int r = min(x, xi);
		int c = min(y, yi);
		if (xi < 0 || xi >= n || yi < 0 || yi >= n||st[xi][yi]==1||g[xi][yi]!=(g[x][y]+1)%k)continue;
		if (i == 1 || i == 5) {
			if (right_to_left[r][c])continue;
		}
		if (i == 7 || i == 3) {
			if (left_to_right[r][c])continue;
		}
		if (i == 1 || i == 5) {
			left_to_right[r][c] = 1;
		}
		else if (i == 7 || i == 3) {
			right_to_left[r][c] = 1;
		}
		s.push_back(number[i]+'0');
		dfs(xi, yi, cur_cnt + 1);
		s.pop_back();
		if (i == 1 || i == 5) {
			left_to_right[r][c] = 0;
		}
		else if (i == 7 || i == 3) {
			right_to_left[r][c] = 0;
		}
	}
	st[x][y] = 0;

}

int main()
{
	cin >> n>>k;
	for (int i = 0; i < n; i++) {
		for (int j = 0; j < n; j++) {
			cin >> g[i][j];
		}
	}
	st[0][0] = 1;
	dfs(0, 0, 1);


	return 0;
}
```

这道题其实思路上不难，只是在正常的dfs搜索路径的基础上，多了个对角线判断而已。

但是恰恰又是这个对角线判断比较难想，在这里的方法是，无论怎么样，只看这两个小方格组成的边长为2的正方形的左上角那个格子，这个格子专门记录这个正方形的对角线情况。

**其实所有的标记和回溯 都放continue后面就好了！！这样不容易出错！不是非要在for循环外面！！！**
