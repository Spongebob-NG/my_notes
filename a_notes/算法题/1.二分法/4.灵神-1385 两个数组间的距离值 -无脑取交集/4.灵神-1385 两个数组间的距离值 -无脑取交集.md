class Solution {
private:
    int lower_bound(vector<int>& nums, int k) {
        int left = 0, right = nums.size() - 1;
        while (left <= right) {
            int mid = (right - left) / 2 + left;
            if (nums[mid] < k)left = mid + 1;
            else right = mid - 1;
        }
        return left;
    }
public:
    int findTheDistanceValue(vector<int>& arr1, vector<int>& arr2, int d) {
        int ans = 0;
        int n1 = arr1.size(); int n2 = arr2.size();
        sort(arr2.begin(), arr2.end());
        for (int i = 0; i < n1; i++) {
            int it = lower_bound(arr2, arr1[i] + d + 1) - 1;
            if (it != -1 && arr2[it] >= arr1[i] - d)continue;
            ans++;
        }
        return ans;
    }
};
