```c++
class Solution {
    // 974. 和可被 K 整除的子数组
    // 由于本题 nums 没有负数，无需调整
    long long subarraysDivByK(vector<int>& nums, int k) {
        unordered_map<int, int> cnt;
        int s = 0;
        long long ans = 0;
        for (int x : nums) {
            cnt[s]++;
            s = (s + x) % k;
            ans += cnt[s];
        }
        return ans;
    }
    
public:
    long long numGoodSubarrays(vector<int>& nums, int k) {
        long long ans = subarraysDivByK(nums, k);
        int n = nums.size();
        int start = 0;
        for (int i = 0; i < n; i++) {
            int x = nums[i];
            if (i < n - 1 && x == nums[i + 1]) {
                continue;
            }
            // 遍历到了连续相同元素段的末尾
            int size = i - start + 1; // 这一段的长度
            for (int sz = 1; sz <= size; sz++) {
                if (1LL * x * sz % k == 0) {
                    // 长为 sz 的子数组元素和能被 k 整除
                    // 一共有 size-sz+1 个长为 sz 的子数组，其中有 size-sz 个重复的
                    ans -= size - sz;
                }
            }
            start = i + 1;
        }
        return ans;
    }
};
```

这是灵神的题解，要注意几点：

1.关于取模的运算，这道题(s[i]-s[j])%k==0，本质上就是说s[i]和s[j]同余，如果两个前缀和的余数相同的话，就能确定一个符合条件的子数组。

2.这种前缀和+哈希的算法题，本质上就是要从题目中找到前缀和之间的关系，列出表达式，再通过表达式的对应关系，通过map来找到符合条件的子数组。

3.！！要更正一下对前缀和的理解，之前的理解是数组下标从1开始存储，然后s[0]判为0，现在这种算法题需要改变一下思路，因为vector的下标必须是从0开始的，所以还是把s[0]判为0，然后对应的nums[i]的前缀和在下标i+1那里。这样的话，计算前缀和就是用s[right+1]-s[left]。为什么一定要把s[0]判为0呢？因为当right和left都为0时，也能符合这个表达式来计算nums[0]的数值。
