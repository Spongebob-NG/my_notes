```c++
#include<iostream>
#include<queue>
#include<vector>
#include<cstring>
using namespace std;
#define N 1005
#define M 10005
#define INF 0x3f
typedef pair<int, int> PII;
int n, m;
int t[N];
int dist[N];
bool f[N];
struct edge {
	int v, w;
};
vector<edge>ver[N];
void dijkstra() {
	priority_queue < PII, vector<PII>, greater<PII>>q;
	dist[1] = t[1];
	q.push({ t[1],1 });
	while (!q.empty()) {
		int time = q.top().first;
		int v = q.top().second;
		q.pop();
		if (f[v])continue;
		f[v] = 1;
		auto x = ver[v].begin();
		while(x!=ver[v].end()){
			int x_v = x->v;
			int x_time = x->w;
			if (dist[x_v] > dist[v] + x_time) {
				dist[x_v] = dist[v] + x_time;
				q.push({ dist[x_v],x_v });
			}
			x++;
		}
	}
}
int main() {
	memset(dist, INF, sizeof(dist));
	cin >> n >> m;
	for (int i = 1; i <= n; i++)cin >> t[i];
	t[1] = 0; t[n] = 0;
	int u, v, c;
	for (int i = 0; i < m; i++) {
		cin >> u >> v >> c;
		ver[u].push_back({ v,c + t[v] });
		ver[v].push_back({ u,c + t[u] });
	}
	dijkstra();
	cout << dist[n] << endl;
}
```



这道题确实是学完之后就直接来用上了，思路是对的，但是有几点需要非常注意：

1.既然这个是无向图，那一条边绝对是要存两次的。虽然说贪心不可能让它走回头路，但是不能保证这条路的起点是谁发起的。就好像3->2->1有条路，但是一开始建图的时候是1->2，如果不存2->1的话，3根本不知道往哪走

2.权值在main建图的时候存过了，就一定不要在dijkstra算法里面再多加一次t[u]啥的，不然时间莫名其妙就变多了
