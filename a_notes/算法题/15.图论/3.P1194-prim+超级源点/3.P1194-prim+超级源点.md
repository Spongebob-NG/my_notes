这道题是最小生成树 需要注意以下问题：

1.INF要设置成0x3f3f3f3f 如果只是0x3f的话虽然memset里面能设置成最大值（它是把每个字节都按照最小的8位来填充的），但是后面的变量设置的只有0x3f。

2.这道题属于多源最短路，因为先买哪个都行，而且题目说了优惠之后的价格**可能比原价高**，所以要弄一个下标为0的作为超级源点，连接每个点，**让每个点都有选择原价购买的权利**。如果不设置一个超级源点的话，后面的只能无脑选所谓的“优惠价”，**其实“优惠价”是比原价要高的**。

3.还有一个思维误区就是，类似prim和dijkstra的算法，老是会觉得一个点vis完了之后就用不上了（我会觉得只能在最新的松弛的点里面选一个放进来）但其实不是这样的，这一轮松弛的点依然可以选择之前的点连接（比如连接超级源点0）这一点要尤其注意

```c++
#include<iostream>
#include<queue>
#include<cstring>
#include<climits>
using namespace std;
#define N 505
#define INF 0x3f3f3f3f
int A, B;
int g[N][N];
int dist[N];
bool vis[N];
int prim() {
	dist[0] = 0;
	int ans = 0;
	for (int i = 0; i <=B; i++) {
		int temp = -1;
		int minn = INT_MAX;
		for (int j = 0; j <= B; j++) {
			if (!vis[j] && dist[j] < minn) {
				minn = dist[j];
				temp = j;
			}
		}
		vis[temp] = 1;
		ans += dist[temp];
		for (int j = 0; j <= B; j++) {
			if (!vis[j] && dist[j] > g[temp][j]) {
				dist[j] = g[temp][j];
			}
		}
	}
	return ans;
}
int main() {
	cin >> A >> B;
	memset(dist, INF, sizeof(dist));
	for (int i = 0; i <= B; i++) {
		g[0][i] = A;
		g[i][0] = A;
	}
	for (int i = 1; i <= B; i++) {
		for (int j = 1; j <= B; j++) {
			int x;
			cin >> x;
			g[i][j] = x;
			g[j][i] = x;
			if (i == j)g[i][j] = INF;
			if (x == 0) {
				g[i][j] = INF;
				g[j][i] = INF;
			}

		}
	}
	int ans=prim();
	cout << ans << endl;

}
```
