这道题 其实就是一个很简单的最小生成树，只不过被题目给的那个所谓的审批规则给吓到了。仔细想想，其实那个审批规则在最小生成树之下肯定是成立的。

但是这道题要注意一个点就是，存图跟平常不一样，这个是直接存x和y的坐标作为城市的坐标。

而且，这种完全图，没有必要所有边都要存进去，因为不是每条边都要在图里面的，会做很多无用功（而且内存也不够，数组太大了 缓存也受不住 命中率很低），只需要在松弛的时候看一下这个点周围的边就好了



```c++
#include<iostream>
#include<cstring>
#include<queue>
#include<vector>
#include<algorithm>
#include<cmath>
#include<stdlib.h>
#include<limits>
using namespace std;
#define INF 0x3f3f3f3f
#define N 5005
typedef pair<double, int> PII;
int n;
bool vis[N];
double dist[N];
struct vertex {
	int x, y;
};
vertex ver[N];
double cal_edge(int i, int j) {
	return sqrt((double)(ver[i].x - ver[j].x) * (ver[i].x - ver[j].x) + (double)(ver[i].y - ver[j].y) * (ver[i].y - ver[j].y));
}
double prim() {
	dist[0] = 0;
	double ans = 0;
	for (int i = 0; i < n; i++) {
		int temp = -1;
		double minn = numeric_limits<double>::max();
		for (int j = 0; j < n; j++) {
			if (!vis[j] && dist[j] < minn) {
				minn = dist[j];
				temp = j;
			}
		}
		if (temp == -1)break;
		ans += dist[temp];
		vis[temp] = 1;
		for (int j = 0; j < n; j++) {
			if (!vis[j] && dist[j] > cal_edge(temp, j)) {
				dist[j] = cal_edge(temp, j);
			}
		}
	}
	return ans;
}
int main() {
	cin >> n;
	int x, y;
	for (int i = 0; i < n; i++)dist[i] = numeric_limits<double>::max();
	for (int i = 0; i < n; i++) {
		cin >> x >> y;
		ver[i].x = x;
		ver[i].y = y;
	}
	double ans = prim();
	printf("%.2f\n", ans);
}
```
