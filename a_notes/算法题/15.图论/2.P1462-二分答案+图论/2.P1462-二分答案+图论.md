```c++
#include<iostream>
#include<queue>
#include<vector>
#include<cstring>
#include<climits>
using namespace std;
int n, m, b;
#define N 10005
#define M 50005
typedef long long ll;
typedef pair<ll, int> PII;
ll deblood[N];
ll fee[N];
ll dist[N];
int f[N];
int ans = INT_MAX;
struct edge {
	int v, deblo;
};
vector<edge>vertex[N];

void dijkstra(int mid) {
	dist[1] = 0;
	priority_queue<PII, vector<PII>, greater<PII>>q;
	q.push({ 0,1 });
	while (!q.empty()) {
		ll deblo = q.top().first;
		int v = q.top().second;
		q.pop();
		if (f[v])continue;
		f[v] = 1;
		for (auto& x : vertex[v]) {
			int x_deblo = x.deblo;
			int x_v = x.v;
			if (fee[x_v] > mid) {
				continue;
			}
			if (dist[x_v] > dist[v] + x_deblo) {
				dist[x_v] = dist[v] + x_deblo;
				q.push({ dist[x_v],x_v });
			}
		}
	}
}
bool check(int mid) {
	if (fee[1] > mid)return false;
	dijkstra(mid);
	if (dist[n] > b)return false;
	else return true;
}
int main() {
	memset(dist, 0x3f, sizeof(dist));
	cin >> n >> m >> b;
	for (int i = 1; i <= n; i++)cin >> fee[i];
	int ai, bi, ci;
	for (int i = 0; i < m; i++) {
		cin >> ai >> bi >> ci;
		vertex[ai].push_back({ bi,ci });
		vertex[bi].push_back({ ai,ci });
	}
	int l = 0, r = 1000000000;
	while (l <= r) {
		int mid = (r - l) / 2 + l;
		memset(dist, 0x3f, sizeof(dist));
		memset(f, 0, sizeof(f));
		if (check(mid)) {
			ans = mid;
			r = mid - 1;
		}
		else {
			l = mid + 1;
		}
	}
	if (ans == INT_MAX)cout << "AFK" << endl;
	else cout << ans << endl;
}
```

这道题其实就是标准的dijkstra板子题 只不过呢配合了二分答案，所以这道题同时也复习了一遍二分答案。

记住，但凡有二分答案的题，都要先满足二分答案的条件，再在这个条件上看能不能满足题目给的其他条件。比如这道题，先看这个城市满不满足小于等于mid，如果满足的话才去看最终血量到底满不满足条件（记得每次都要初始化一下数组）
