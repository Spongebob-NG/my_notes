```c++
class Solution {
public:
    bool isValid(string s) {
        stack<char>myStack;
        for (int i = 0; i < s.length(); i++) {
            char x;
            if (!myStack.empty())x = myStack.top();
            if (!myStack.empty()&&((s[i] == ')' && x == '(') || (s[i] == ']' && x == '[') || (s[i] == '}' && x == '{'))) {
                myStack.pop();
            }
            else myStack.push(s[i]);

        }
        if (myStack.empty())return true;
        return false;

    }
}; 
```
