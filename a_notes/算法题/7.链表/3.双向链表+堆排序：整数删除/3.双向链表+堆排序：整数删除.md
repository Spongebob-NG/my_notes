这种涉及到在中间频繁删除的东西 还得是数组模拟双向链表（并且要弄一个头结点作为哨兵节点，且最后遍历的时候一定要从这个哨兵节点开始，不然的话会直接打印出来已经被删除的元素）

```c++
#include <iostream>
#include<queue>
using namespace std;
typedef long long ll;
typedef pair<ll, int> PII;
const int N = 5e5 + 10;
ll a[N];
int pre[N];
int ne[N];
bool del[N];
int n, k;
int main()
{
  ios::sync_with_stdio(false);
  cin.tie(NULL);
	cin >> n >> k;
	priority_queue<PII,vector<PII>,greater<PII>>q;
	ne[0] = 1;
	for (int i = 1; i <= n; i++) {
		cin >> a[i];
		pre[i] = i - 1;
		ne[i] = i + 1;
		q.push({ a[i],i });
	}
	while (k--) {
		ll x = q.top().first; int index = q.top().second;
		q.pop();
		while (del[index]||x!=a[index]) {
			x = q.top().first;
			index = q.top().second;
			q.pop();
		}
		if (pre[index] != 0) { 
			a[pre[index]] += x; 
			q.push({ a[pre[index]],pre[index] });
		}
		if (ne[index] != n + 1) {
			a[ne[index]] += x;
			q.push({ a[ne[index]],ne[index] });
		}
		del[index] = true;
		ne[pre[index]] = ne[index];
		pre[ne[index]] = pre[index];
	}
	for (int i = ne[0]; i!=n+1; i = ne[i])cout << a[i] << ' ';

	return 0;
}
```
